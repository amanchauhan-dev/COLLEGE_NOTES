## Q.1 (a) What do you mean by responsive design? Explain with example. [03]

=> **Responsive design** makes a web page adjust automatically according to screen size such as mobile, tablet and desktop.

```html
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<style>
.box { width: 50%; background: lightblue; }
@media (max-width: 600px) {
  .box { width: 100%; }
}
</style>
<div class="box">Responsive Box</div>
```

## Q.1 (b) Explain asynchronous programming in detail. [04]

=> **Asynchronous programming** allows a task to run without blocking the execution of the next statement.

=> It is useful for file handling, network requests, timers and database operations.

```js
console.log("Start");
setTimeout(() => console.log("After 2 seconds"), 2000);
console.log("End");
```

=> Output is `Start`, `End`, then `After 2 seconds`.

## Q.1 (c) Explain steps to implement AngularJS routing with example. [07]

=> Routing is used to show different views in a single page application.

### Steps

1. Include AngularJS and `angular-route.js`.
2. Create module with `ngRoute`.
3. Configure routes using `$routeProvider`.
4. Use links with `#!`.
5. Add `<div ng-view></div>`.

```html
<html ng-app="app">
<body>
<a href="#!/">Home</a>
<a href="#!/books">Books</a>
<div ng-view></div>

<script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>
<script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular-route.js"></script>
<script>
var app = angular.module("app", ["ngRoute"]);
app.config(function($routeProvider) {
  $routeProvider
    .when("/", { template: "<h2>Home</h2>" })
    .when("/books", { template: "<h2>Books</h2>" })
    .otherwise({ redirectTo: "/" });
});
</script>
</body>
</html>
```

## Q.3 (c OR) Print query string Email and Mobile. [07]

```js
const http = require("http");
const url = require("url");

http.createServer((req, res) => {
  const q = url.parse(req.url, true).query;
  res.writeHead(200, {"Content-Type": "text/html"});
  res.write("Email: " + q.Email + "<br>");
  res.write("Mobile: " + q.Mobile);
  res.end();
}).listen(8080);
```

## Q.4 (c) Create server that logs visited pages. [07]

```js
const http = require("http");
const fs = require("fs");

http.createServer((req, res) => {
  const log = new Date().toISOString() + " " + req.url + "\n";
  fs.appendFile("log.txt", log, err => {
    if (err) throw err;
  });
  res.end("Page visited: " + req.url);
}).listen(8080);
```

## Q.5 (c) MongoDB insert and delete operations. [07]

```js
const { MongoClient } = require("mongodb");

async function main() {
  const client = new MongoClient("mongodb://127.0.0.1:27017");
  await client.connect();
  const col = client.db("college").collection("students");

  await col.insertOne({ rollno: 1, name: "Aman" });
  await col.insertMany([{ rollno: 2, name: "Ravi" }, { rollno: 3, name: "Neha" }]);
  await col.deleteOne({ rollno: 1 });
  await col.deleteMany({ name: "Ravi" });

  await client.close();
}
main();
```

## Q.5 (c OR) Query Student collection in GTU database. [07]

```js
const { MongoClient } = require("mongodb");

async function main() {
  const client = new MongoClient("mongodb://127.0.0.1:27017");
  await client.connect();
  const students = client.db("GTU").collection("Student");

  const ageResult = await students.find({ age: { $gte: 20, $lte: 22 } }).toArray();
  const awpResult = await students.find({ course: "AWP" }).toArray();

  console.log(ageResult);
  console.log(awpResult);
  await client.close();
}
main();
```
