## Q.1 (a) Define CSS. Enlist benefits. [03]

=> **CSS** stands for **Cascading Style Sheets**.

=> CSS is used to style HTML pages by controlling color, font, spacing, border, layout and responsiveness.

### Benefits

1. Separates content from design.
2. Saves time by reusing styles.
3. Improves consistency across pages.
4. Makes website maintenance easy.
5. Supports responsive design.

## Q.1 (b) Explain CSS types with suitable examples. [04]

### 1. Inline CSS

```html
<p style="color:red">Inline CSS</p>
```

=> Written inside HTML tag using `style` attribute.

### 2. Internal CSS

```html
<style>
p {
  color: blue;
}
</style>
```

=> Written inside `<style>` tag in the same HTML page.

### 3. External CSS

```html
<link rel="stylesheet" href="style.css">
```

=> Written in separate `.css` file. It is best for large websites.

## Q.3 (c) Custom filter to capitalize vowels and read `SampleText.txt`. [07]

```html
<!DOCTYPE html>
<html ng-app="app">
<body ng-controller="Ctrl">
  <h3>Custom Filter Example</h3>
  <p>{{text | vowelCaps}}</p>

  <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>
  <script>
  var app = angular.module("app", []);

  app.filter("vowelCaps", function() {
    return function(input) {
      return (input || "").toLowerCase().replace(/[aeiou]/g, function(ch) {
        return ch.toUpperCase();
      });
    };
  });

  app.controller("Ctrl", function($scope, $http) {
    $http.get("SampleText.txt").then(function(response) {
      $scope.text = response.data;
    });
  });
  </script>
</body>
</html>
```

=> Custom filter changes vowels to uppercase and remaining characters to lowercase.

## Q.4 (c) Explain input field states in AngularJS. [07]

| State | Meaning |
|---|---|
| `$untouched` | Field not visited |
| `$touched` | Field visited and lost focus |
| `$pristine` | Field not changed |
| `$dirty` | Field changed |
| `$valid` | Input is valid |
| `$invalid` | Input is invalid |

```html
<form name="f" ng-app="">
  <input name="email" type="email" ng-model="email" required>
  <p ng-show="f.email.$touched && f.email.$invalid">
    Invalid email
  </p>
  <p>Valid: {{f.email.$valid}}</p>
  <p>Dirty: {{f.email.$dirty}}</p>
</form>
```

=> These states help display proper validation messages.

## Q.4 (c OR) Server that writes visited page log. [07]

```js
const http = require("http");
const fs = require("fs");

http.createServer(function(req, res) {
  const log = new Date().toISOString() + " " + req.url + "\n";

  fs.appendFile("log.txt", log, function(err) {
    if (err) throw err;
    res.end("Logged page: " + req.url);
  });
}).listen(8080);
```

=> `fs.appendFile()` adds each visited URL into `log.txt`.

## Q.5 (b) Explain piping and chaining of streams. [04]

=> Piping connects output of one stream to input of another stream.

=> Chaining means connecting multiple stream operations together.

```js
const fs = require("fs");
const zlib = require("zlib");

fs.createReadStream("input.txt")
  .pipe(zlib.createGzip())
  .pipe(fs.createWriteStream("input.txt.gz"));
```

=> This reads file, compresses it and writes compressed output.

## Q.5 (c) Update and delete Employee database using MongoDB. [07]

```js
const { MongoClient } = require("mongodb");

async function main() {
  const client = new MongoClient("mongodb://127.0.0.1:27017");

  try {
    await client.connect();
    const emp = client.db("company").collection("employee");

    await emp.updateOne(
      { id: 1 },
      { $set: { department: "IT", salary: 50000 } }
    );

    await emp.deleteOne({ id: 2 });
    console.log("Employee updated and deleted");
  } catch (err) {
    console.log(err.message);
  } finally {
    await client.close();
  }
}

main();
```

  console.log(data);
});
