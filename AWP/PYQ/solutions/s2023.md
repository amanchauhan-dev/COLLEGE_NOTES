## Q.2 (b) Define MVC and implementation in AngularJS. [04]

=> MVC means Model View Controller.

```html
<div ng-app="app" ng-controller="Ctrl">
  <input ng-model="name">
  <p>{{name}}</p>
</div>
<script>
angular.module("app", []).controller("Ctrl", function($scope) {
  $scope.name = "Aman";
});
</script>
```

=> `name` is model, HTML is view and `Ctrl` is controller.

## Q.2 (c OR) Bootstrap student form with email validation. [07]

```html
<link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/css/bootstrap.min.css">

  let email = document.getElementById("email").value;

    alert("Invalid Email");
    return;

    document.getElementById("en").value + " " +
    document.getElementById("name").value + " " + email;
}
</script>
```

## Q.3 (c) Custom filter to capitalize vowels and read SampleText.txt. [07]

```html
<div ng-app="app" ng-controller="Ctrl">
  <p>{{text | vowelCaps}}</p>
</div>
<script>
var app = angular.module("app", []);
app.filter("vowelCaps", function() {
  return function(input) {
    return (input || "").toLowerCase().replace(/[aeiou]/g, ch => ch.toUpperCase());
  };
});
app.controller("Ctrl", function($scope, $http) {
  $http.get("SampleText.txt").then(res => $scope.text = res.data);
});
</script>
```

## Q.4 (c OR) Server that writes visited page log. [07]

```js
const http = require("http");
const fs = require("fs");
http.createServer((req, res) => {
  fs.appendFile("log.txt", req.url + "\n", err => {
    if (err) throw err;
  });
  res.end("Logged " + req.url);
}).listen(8080);
```

## Q.5 (c) Update and delete Employee database using MongoDB. [07]

```js
const { MongoClient } = require("mongodb");
async function main() {
  const client = new MongoClient("mongodb://127.0.0.1:27017");
  await client.connect();
  const emp = client.db("company").collection("employee");
  await emp.updateOne({ id: 1 }, { $set: { department: "IT" } });
  await emp.deleteOne({ id: 2 });
  await client.close();
}
main();
```
